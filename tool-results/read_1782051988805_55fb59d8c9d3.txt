   373→function OverviewTab({
   374→  childList,
   375→  parents,
   376→  onSelectChild,
   377→  onGiveTokens,
   378→  onLogSpend,
   379→}: {
   380→  childList: Child[];
   381→  parents: { id: string; name: string; role: string; avatarColor: string; avatarPhoto?: string }[];
   382→  onSelectChild: (c: Child) => void;
   383→  onGiveTokens: (c: Child) => void;
   384→  onLogSpend: (owner: SpendingOwner) => void;
   385→}) {
   386→  const totalSavings = useStore(familyTotalSavings);
   387→  const totalInvested = useStore(familyTotalInvested);
   388→  const thisMonth = useStore(familyThisMonthSaved);
   389→  const totalGoals = useStore(familyTotalGoals);
   390→  const tokens = useStore(parentTokenBalance);
   391→  // Raw spending state — derive breakdown via useMemo to avoid infinite re-renders
   392→  // (familySpendingBreakdown selector returns a new array each call).
   393→  const allSpending = useStore((s) => s.spending);
   394→  const familySpending = useMemo(
   395→    () =>
   396→      allSpending
   397→        .filter((e) => {
   398→          const d = new Date(e.timestamp);
   399→          const now = new Date();
   400→          return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth();
   401→        })
   402→        .reduce((sum, e) => sum + e.amount, 0),
   403→    [allSpending]
   404→  );
   405→  const spendingBreakdown = useMemo(() => {
   406→    const now = new Date();
   407→    const map = new Map<string, { ownerKind: "parent" | "child"; ownerName: string; total: number }>();
   408→    for (const e of allSpending) {
   409→      const d = new Date(e.timestamp);
   410→      if (d.getFullYear() !== now.getFullYear() || d.getMonth() !== now.getMonth()) continue;
   411→      const existing = map.get(e.ownerId);
   412→      if (existing) {
   413→        existing.total += e.amount;
   414→      } else {
   415→        map.set(e.ownerId, { ownerKind: e.ownerKind, ownerName: e.ownerName, total: e.amount });
   416→      }
   417→    }
   418→    return Array.from(map.entries())
   419→      .map(([ownerId, v]) => ({ ownerId, ...v }))
   420→      .sort((a, b) => b.total - a.total);
   421→  }, [allSpending]);
   422→  // Raw state, derive recent slice via useMemo.
   423→  const allTransactions = useStore((s) => s.transactions);
   424→  const transactions = useMemo(
   425→    () => allTransactions.slice(0, 8),
   426→    [allTransactions]
   427→  );
   428→  const children = childList;
   429→
   430→  const kpis = [
   431→    {
   432→      label: "Family Savings",
   433→      value: formatUGX(totalSavings),
   434→      sub: `of ${formatUGX(totalGoals)} goal`,
   435→      pct: totalGoals > 0 ? (totalSavings / totalGoals) * 100 : 0,
   436→      icon: PiggyBank,
   437→      hero: true,
   438→    },
   439→    {
   440→      label: "Investments",
   441→      value: formatUGX(totalInvested),
   442→      sub: "active portfolio",
   443→      icon: TrendingUp,
   444→    },
   445→    {
   446→      label: "This Month Saved",
   447→      value: formatUGX(thisMonth),
   448→      sub: "credits this period",
   449→      icon: Calendar,
   450→    },
   451→    {
   452→      label: "Tokens in Circulation",
   453→      value: `${tokens} ◈`,
   454→      sub: `cost UGX ${formatUGXPlain(tokens * TOKEN_BUY_RATE)}`,
   455→      icon: Sparkles,
   456→    },
   457→  ];
   458→
   459→  return (
   460→    <div className="space-y-8 animate-fade-up">
   461→      {/* KPI grid */}
   462→      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4 stagger">
   463→        {kpis.map((k) => (
   464→          <div
   465→            key={k.label}
   466→            className={`rounded-lg p-6 ${k.hero ? "surface-wood-strong" : "surface-wood card-hover"}`}
   467→          >
   468→            <div className="flex items-center justify-between mb-4">
   469→              <div className="micro-label">{k.label}</div>
   470→              <k.icon className="h-3.5 w-3.5" style={{ color: "var(--primary)" }} />
   471→            </div>
   472→            <div
   473→              className={`font-editorial tabular-nums leading-none ${
   474→                k.hero ? "text-3xl text-gold-foil" : "text-2xl text-foreground"
   475→              }`}
   476→            >
   477→              {k.value}
   478→            </div>
   479→            <div className="text-[10px] text-foreground/45 mt-2">{k.sub}</div>
   480→            {k.pct !== undefined && (
   481→              <div className="progress-thin mt-3">
   482→                <div style={{ width: `${Math.min(100, k.pct)}%` }} />
   483→              </div>
   484→            )}
   485→          </div>
   486→        ))}
   487→      </div>
   488→
   489→      {/* Visualizations section — editorial SVG charts.
   490→          Parents understand graphs better than ledger tables. */}
   491→      <div className="space-y-3 mb-2">
   492→        <div className="flex items-center justify-between px-1">
   493→          <div>
   494→            <div className="micro-label-gold mb-1">Visualizations</div>
   495→            <h2 className="font-editorial text-lg text-foreground tracking-wide">
   496→              Family Insights
   497→            </h2>
   498→          </div>
   499→          <span className="micro-label">last 6 months</span>
   500→        </div>
   501→      </div>
   502→
   503→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4 mb-8 stagger">
   504→        <SavingsTrendChart />
   505→        <DistributionDonut childList={children} />
   506→        <CashFlowBars />
   507→        <GoalRadials childList={children} />
   508→      </div>
