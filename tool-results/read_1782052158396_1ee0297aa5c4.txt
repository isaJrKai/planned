   430→function SpendingTab({ child, onSpend }: { child: Child; onSpend: () => void }) {
   431→  // Subscribe to RAW state — never derive arrays in the selector (causes
   432→  // infinite re-renders because each call produces new object references).
   433→  const categories = useStore((s) => s.categories);
   434→  const spending = useStore((s) => s.spending);
   435→
   436→  // Derive per-child category breakdown with useMemo.
   437→  const childCategories = useMemo(() => {
   438→    const monthStart = new Date();
   439→    monthStart.setDate(1);
   440→    monthStart.setHours(0, 0, 0, 0);
   441→    const monthStartTime = monthStart.getTime();
   442→
   443→    return categories.map((category) => {
   444→      const spent = spending
   445→        .filter(
   446→          (e) =>
   447→            e.ownerId === child.id &&
   448→            e.category === category.name &&
   449→            e.timestamp >= monthStartTime
   450→        )
   451→        .reduce((sum, e) => sum + e.amount, 0);
   452→      const remaining = Math.max(0, category.budget - spent);
   453→      const pct = category.budget > 0 ? Math.min(100, (spent / category.budget) * 100) : 0;
   454→      return { category, spent, remaining, pct };
   455→    });
   456→  }, [categories, spending, child.id]);
   457→
   458→  const childSpending = useMemo(
   459→    () =>
   460→      spending
   461→        .filter((e) => e.ownerId === child.id)
   462→        .sort((a, b) => b.timestamp - a.timestamp),
   463→    [spending, child.id]
   464→  );
   465→
   466→  const totalSpent = childSpending
   467→    .filter((e) => new Date(e.timestamp).getMonth() === new Date().getMonth())
   468→    .reduce((sum, e) => sum + e.amount, 0);
   469→
   470→  const totalBudget = childCategories.reduce((sum, c) => sum + c.category.budget, 0);
   471→
   472→  return (
   473→    <div className="space-y-8 animate-fade-up">
   474→      <div className="flex items-end justify-between">
   475→        <div>
   476→          <div className="micro-label-gold mb-2">Monthly Budget</div>
   477→          <div className="font-editorial text-3xl text-gold-foil-static tabular-nums">
   478→            {formatUGX(totalSpent)}
   479→          </div>
   480→          <div className="micro-label mt-1">of {formatUGX(totalBudget)} allocated</div>
   481→        </div>
   482→        <button
   483→          onClick={onSpend}
   484→          className="btn-gold px-4 py-2 rounded text-xs tracking-wider flex items-center gap-2"
   485→        >
   486→          <Plus className="h-3 w-3" /> Log Spend
   487→        </button>
   488→      </div>
   489→
   490→      <div className="grid md:grid-cols-2 gap-4">
   491→        {childCategories.map(({ category, spent, remaining, pct }) => (
   492→          <div key={category.id} className="surface-wood rounded-lg p-5">
   493→            <div className="flex items-center justify-between mb-3">
   494→              <div className="font-editorial text-sm tracking-wide text-foreground">
   495→                {category.name}
   496→              </div>
   497→              <span className={`pill ${pct >= 100 ? "pill-amber" : pct >= 75 ? "pill-amber" : "pill-green"}`}>
   498→                {pct.toFixed(0)}%
   499→              </span>
   500→            </div>
   501→            <div className="progress-thin mb-3">
   502→              <div style={{ width: `${Math.min(100, pct)}%` }} />
   503→            </div>
   504→            <div className="flex justify-between text-xs">
   505→              <span className="micro-label">Spent</span>
   506→              <span className="font-editorial tabular-nums text-foreground/80">
   507→                {formatUGX(spent)}
   508→              </span>
   509→            </div>
   510→            <div className="flex justify-between text-xs mt-1">
   511→              <span className="micro-label">Remaining</span>
   512→              <span
   513→                className="font-editorial tabular-nums"
   514→                style={{ color: remaining > 0 ? "#E8D5A0" : "#D4943A" }}
   515→              >
   516→                {formatUGX(remaining)}
   517→              </span>
   518→            </div>
   519→          </div>
   520→        ))}
   521→      </div>
   522→
   523→      <div>
   524→        <h3 className="font-editorial text-lg tracking-wide text-foreground mb-4">
   525→          Spending Log
   526→        </h3>
   527→        <div className="surface-wood rounded-lg overflow-hidden">
   528→          <table className="ledger-table">
   529→            <thead>
   530→              <tr>
   531→                <th>Date</th>
   532→                <th>Category</th>

   568→// ---- Worksheet Tab ---------------------------------------------------------
   569→// Bug #11 fix: totalReceived filters to THIS MONTH only — previously it summed
   570→// all-time credits which made it mismatch totalSpent (which was month-only).
   571→
   572→function WorksheetTab({ child }: { child: Child }) {
   573→  const now = new Date();
   574→  const monthLabel = now.toLocaleDateString("en-GB", {
   575→    month: "long",
   576→    year: "numeric",
   577→  });
   578→
   579→  // Raw state subscriptions (avoid derived arrays in selectors).
   580→  const allTransactions = useStore((s) => s.transactions);
   581→  const allSpending = useStore((s) => s.spending);
   582→
   583→  const monthStart = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
   584→
   585→  const transactions = useMemo(
   586→    () => allTransactions.filter((t) => t.childId === child.id),
   587→    [allTransactions, child.id]
   588→  );
   589→  const spending = useMemo(
   590→    () => allSpending.filter((e) => e.ownerId === child.id),
   591→    [allSpending, child.id]
   592→  );
   593→
   594→  const receivedThisMonth = transactions
   595→    .filter(
   596→      (t) =>
   597→        t.timestamp >= monthStart &&
   598→        (t.type === "save" || t.type === "redeem")
   599→    )
   600→    .reduce((sum, t) => sum + t.amount, 0);
   601→
   602→  const spentThisMonth = spending
   603→    .filter((e) => e.timestamp >= monthStart)
   604→    .reduce((sum, e) => sum + e.amount, 0);
   605→
   606→  const net = receivedThisMonth - spentThisMonth;
   607→
   608→  const rows = [
   609→    { label: "Savings Deposits (this month)", value: receivedThisMonth, type: "credit" },
   610→    { label: "Spending (this month)", value: spentThisMonth, type: "debit" },
   611→    { label: "Net Cash Flow", value: net, type: net >= 0 ? "credit" : "debit" },
   612→  ];
   613→
   614→  return (
   615→    <div className="space-y-8 animate-fade-up">
   616→      <div>
   617→        <div className="micro-label-gold mb-2">Monthly Worksheet</div>
   618→        <h2 className="font-editorial text-3xl text-foreground tracking-wide">
   619→          {monthLabel}
   620→        </h2>
   621→      </div>
   622→
   623→      <div className="surface-wood-strong rounded-lg p-6">
   624→        <div className="micro-label mb-4">Cash Flow Summary</div>
   625→        <div className="divider-gold mb-5" />
   626→        <div className="space-y-4">
   627→          {rows.map((r) => (
   628→            <div key={r.label} className="flex items-center justify-between">
   629→              <span className="text-sm text-foreground/70 tracking-wide">{r.label}</span>
   630→              <span
   631→                className="font-editorial text-xl tabular-nums"
   632→                style={{
   633→                  color:
   634→                    r.type === "credit" ? "#6BBF8A" : "#D4943A",
   635→                }}
   636→              >
   637→                {r.type === "credit" ? "+" : "−"}
   638→                {formatUGXPlain(Math.abs(r.value))}
   639→              </span>
   640→            </div>
   641→          ))}
   642→        </div>
   643→        <div className="divider-gold my-5" />
   644→        <div className="flex items-center justify-between">
   645→          <span className="micro-label-gold">Net Position</span>
   646→          <span
   647→            className="font-editorial text-2xl text-gold-foil-static tabular-nums"
   648→          >
   649→            {net >= 0 ? "+" : "−"}
   650→            {formatUGXPlain(Math.abs(net))}
   651→          </span>
   652→        </div>
   653→      </div>
   654→
   655→      <div className="grid md:grid-cols-2 gap-4">
   656→        <div className="surface-wood rounded-lg p-5">
   657→          <div className="flex items-center gap-2 mb-3">
   658→            <Target className="h-3.5 w-3.5" style={{ color: "var(--primary)" }} />
   659→            <span className="micro-label-gold">Goal Progress</span>
   660→          </div>
   661→          <div className="font-editorial text-2xl tabular-nums text-foreground">
   662→            {formatUGX(child.currentAmount)}
   663→          </div>
   664→          <div className="micro-label mt-1">of {formatUGX(child.goalAmount)}</div>
   665→          <div className="progress-thin mt-3">
   666→            <div
   667→              style={{
   668→                width: `${Math.min(100, (child.currentAmount / child.goalAmount) * 100)}%`,
   669→              }}
   670→            />
   671→          </div>
   672→        </div>
   673→        <div className="surface-wood rounded-lg p-5">
   674→          <div className="flex items-center gap-2 mb-3">
   675→            <Calendar className="h-3.5 w-3.5" style={{ color: "var(--primary)" }} />
   676→            <span className="micro-label-gold">Daily Save Avg</span>
   677→          </div>
   678→          <div className="font-editorial text-2xl tabular-nums text-foreground">
   679→            {formatUGX(receivedThisMonth / Math.max(1, now.getDate()))}
   680→          </div>
   681→          <div className="micro-label mt-1">this month, per day</div>
   682→        </div>
   683→      </div>
   684→    </div>
   685→  );
   686→}

   690→function InvestmentsTab({
   691→  child,
   692→  onSelect,
   693→}: {
   694→  child: Child;
   695→  onSelect: (i: Investment) => void;
   696→}) {
   697→  const allInvestments = useStore((s) => s.investments);
   698→  const investments = useMemo(
   699→    () => allInvestments.filter((i) => i.childId === child.id),
   700→    [allInvestments, child.id]
   701→  );
   702→  const totalValue = investments
   703→    .filter((i) => i.status === "active")
   704→    .reduce((sum, i) => sum + i.currentValue, 0);
   705→  const totalPrincipal = investments
   706→    .filter((i) => i.status === "active")
   707→    .reduce((sum, i) => sum + i.amountInvested, 0);
   708→  const gain = totalValue - totalPrincipal;
   709→
   710→  return (
   711→    <div className="space-y-8 animate-fade-up">
   712→      <div className="surface-wood-strong rounded-lg p-6">
   713→        <div className="micro-label-gold mb-2">Portfolio Value</div>
   714→        <div className="font-editorial text-4xl text-gold-foil tabular-nums">
   715→          {formatUGX(totalValue)}
   716→        </div>
   717→        <div className="mt-3 flex items-center gap-3">
   718→          <span
   719→            className="text-sm tabular-nums"
   720→            style={{ color: gain >= 0 ? "#6BBF8A" : "#D4943A" }}
   721→          >
   722→            {gain >= 0 ? "+" : "−"}
   723→            {formatUGXPlain(Math.abs(gain))}
   724→          </span>
   725→          <span className="micro-label">all-time gain</span>
   726→        </div>
   727→      </div>
   728→
   729→      <div className="surface-wood rounded-lg overflow-hidden">
   730→        <table className="ledger-table">
   731→          <thead>
   732→            <tr>
   733→              <th>Position</th>
   734→              <th>Type</th>
   735→              <th className="text-right">Principal</th>
   736→              <th className="text-right">Value</th>
   737→              <th className="text-right">Gain</th>
   738→            </tr>
   739→          </thead>
   740→          <tbody>
   741→            {investments.map((inv) => {
   742→              const g = inv.currentValue - inv.amountInvested;
   743→              const pct = inv.amountInvested > 0 ? (g / inv.amountInvested) * 100 : 0;
   744→              return (
   745→                <tr
   746→                  key={inv.id}
   747→                  onClick={() => onSelect(inv)}
   748→                  className="cursor-pointer"
   749→                >
   750→                  <td className="font-editorial tracking-wide text-foreground">
   751→                    {inv.name}
   752→                  </td>
   753→                  <td className="text-foreground/60 text-xs">{inv.type}</td>
   754→                  <td className="text-right tabular-nums text-foreground/70">
   755→                    {formatUGXPlain(inv.amountInvested)}
   756→                  </td>
   757→                  <td className="text-right tabular-nums text-foreground">
   758→                    {formatUGXPlain(inv.currentValue)}
   759→                  </td>
   760→                  <td
   761→                    className="text-right tabular-nums"
   762→                    style={{ color: g >= 0 ? "#6BBF8A" : "#D4943A" }}
   763→                  >
   764→                    {g >= 0 ? "+" : ""}
   765→                    {pct.toFixed(1)}%
   766→                  </td>
   767→                </tr>
   768→              );
   769→            })}
   770→            {investments.length === 0 && (
   771→              <tr>
   772→                <td colSpan={5} className="text-center text-foreground/40 py-8">
   773→                  No investments yet.
   774→                </td>
   775→              </tr>
   776→            )}
   777→          </tbody>
   778→        </table>
   779→      </div>
   780→    </div>
   781→  );
   782→}
