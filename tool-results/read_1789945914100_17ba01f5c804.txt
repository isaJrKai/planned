    14→function useAction() {
    15→  const router = useRouter();
    16→  const [busy, setBusy] = React.useState(false);
    17→  const [err, setErr] = React.useState<ApiErr | null>(null);
    18→  const [msg, setMsg] = React.useState<string | null>(null);
    19→  const run = async <T,>(fn: () => Promise<{ ok: true; data: T } | { ok: false; error: ApiErr }>, onOk?: (d: T) => void) => {
    20→    setBusy(true); setErr(null); setMsg(null);
    21→    const r = await fn();
    22→    setBusy(false);
    23→    if (r.ok) { onOk?.(r.data); router.refresh(); return true; }
    24→    setErr(r.error); return false;
    25→  };
    26→  return { busy, err, msg, setMsg, setErr, run };
    27→}

   149→export function AllowanceButton({ currency, members, goals, first }: { currency: string; members: Member[]; goals: GoalLite[]; first?: boolean }) {
   150→  const [open, setOpen] = React.useState(false);
   151→  const [cadence, setCadence] = React.useState<"weekly" | "monthly">("weekly");
   152→  const [memberId, setMemberId] = React.useState(members.find((m) => m.role === "child")?.id ?? "");
   153→  const [save, setSave] = React.useState(20);
   154→  const [goalPct, setGoalPct] = React.useState(0);
   155→  const a = useAction();
   156→  const kids = members.filter((m) => m.role === "child");
   157→  const memberGoals = goals.filter((g) => g.memberId === memberId);
   158→  async function submit(e: React.FormEvent<HTMLFormElement>) {
   159→    e.preventDefault();
   160→    const fd = new FormData(e.currentTarget);
   161→    const amount = parseMoneyInput(String(fd.get("amount")), currency) ?? 0;
   162→    if (!amount) { a.setErr({ code: "v", message: "Enter a valid amount." }); return; }
   163→    const goalId = fd.get("goalId") ? String(fd.get("goalId")) : null;
   164→    const ok = await a.run(() => api("/api/allowances", { body: { memberId, amount, cadence, dayOfPeriod: Number(fd.get("day")), savePercent: save, goalPercent: goalId ? goalPct : 0, goalId } }));
   165→    if (ok) setOpen(false);
   166→  }
   167→  if (!kids.length) return null;
   168→  const spend = Math.max(0, 100 - save - (memberGoals.length ? goalPct : 0));
   169→  return (
   170→    <>
   171→      <Button variant={first ? "primary" : "secondary"} onClick={() => setOpen(true)}><Plus className="h-4 w-4" />New allowance</Button>
   172→      <Modal open={open} onClose={() => setOpen(false)} title="Set up an allowance">
   173→        <form onSubmit={submit} className="space-y-4" noValidate>
   174→          {a.err && <Notice tone="spend">{a.err.message}</Notice>}
   175→          <Field label="For" htmlFor="al-m"><Select id="al-m" value={memberId} onChange={(e) => setMemberId(e.target.value)}>{kids.map((k) => <option key={k.id} value={k.id}>{k.name}</option>)}</Select></Field>
   176→          <Field label="Amount" htmlFor="al-amt"><MoneyInput currency={currency} id="al-amt" name="amount" /></Field>
   177→          <div className="grid grid-cols-2 gap-3">
   178→            <Field label="How often" htmlFor="al-c"><Select id="al-c" value={cadence} onChange={(e) => setCadence(e.target.value as "weekly" | "monthly")}><option value="weekly">Every week</option><option value="monthly">Every month</option></Select></Field>
   179→            <Field label={cadence === "weekly" ? "On" : "On day"} htmlFor="al-d">
   180→              <Select id="al-d" name="day" defaultValue={cadence === "weekly" ? 6 : 1}>
   181→                {cadence === "weekly" ? WEEKDAYS.map((d, i) => <option key={d} value={i}>{d}</option>) : Array.from({ length: 28 }, (_, i) => <option key={i + 1} value={i + 1}>{i + 1}</option>)}
   182→              </Select>
   183→            </Field>
   184→          </div>
   185→          <fieldset className="border border-line rounded-md p-4 space-y-3">
   186→            <legend className="px-1 text-[13.5px] font-medium text-ink-2">How it&rsquo;s split</legend>
   187→            <label className="block text-[14px]"><span className="flex justify-between"><span>To savings</span><span className="tnum text-ink-2">{save}%</span></span><input type="range" min={0} max={100 - goalPct} value={save} onChange={(e) => setSave(Number(e.target.value))} className="w-full accent-[#1a8a4a]" aria-label="Percent to savings" /></label>
   188→            {memberGoals.length > 0 && (
   189→              <>
   190→                <Field label="Toward a goal" htmlFor="al-g"><Select id="al-g" name="goalId" defaultValue=""><option value="">No goal</option>{memberGoals.map((g) => <option key={g.id} value={g.id}>{g.name}</option>)}</Select></Field>
   191→                <label className="block text-[14px]"><span className="flex justify-between"><span>To that goal</span><span className="tnum text-ink-2">{goalPct}%</span></span><input type="range" min={0} max={100 - save} value={goalPct} onChange={(e) => setGoalPct(Number(e.target.value))} className="w-full accent-[#c47a09]" aria-label="Percent to goal" /></label>
   192→              </>
   193→            )}
   194→            <p className="text-[13.5px] text-ink-2">Spending gets the rest: <strong className="text-ink tnum">{spend}%</strong>.</p>
   195→          </fieldset>
   196→          <p className="text-[13px] text-ink-3">Paid automatically from the family pool on the day you chose. You can also pay early from the Allowances page — never twice in one period.</p>
   197→          <div className="flex justify-end gap-2"><Button type="button" variant="ghost" onClick={() => setOpen(false)}>Cancel</Button><Button type="submit" loading={a.busy}>Create</Button></div>
   198→        </form>
   199→      </Modal>
   200→    </>
   201→  );
   202→}
