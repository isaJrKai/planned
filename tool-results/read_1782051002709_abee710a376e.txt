     2→     2→// AUTH LIBRARY — JWT session, password hashing, TOTP 2FA, role checks, audit
     3→     3→// ============================================================================
     4→     4→// This is the single source of truth for authentication and authorization
     5→     5→// in Planned. The session is a JWT signed with jose, stored in an HttpOnly
     6→     6→// cookie. Server-side role checks happen here — client-side checks are
     7→     7→// decorative only and never trusted.
     8→     8→//
     9→     9→// Founder-only administration:
    10→    10→//   - Exactly one SUPER_ADMIN exists (enforced at /setup time)
    11→    11→//   - All other accounts default to USER — NEVER trust client-supplied role
    12→    12→//   - /admin/*, /system/*, /analytics/*, /feature-flags/*, /security/*
    13→    13→//     require role === "SUPER_ADMIN"
    14→    14→//   - Every access attempt is logged to AuditLog
    15→    15→//   - TOTP 2FA optional at enrollment, required for sensitive ops once on
    16→    16→//
    17→    17→// Authority principle:
    18→    18→//   - superAdminUserId is AUTHORITATIVE in SystemSettings
    19→    19→//   - Email is metadata only — can drift, never used for access checks

    95→    95→// ---- Password policy -------------------------------------------------------
    96→    96→// Enforced server-side everywhere a password is accepted. Client-side checks
    97→    97→// are decorative — this is the authoritative gate.
    98→    98→//   - minimum 12 characters
    99→    99→//   - at least one uppercase letter
   100→   100→//   - at least one lowercase letter
   101→   101→//   - at least one digit
   102→   102→// (Symbols recommended but not required — they're often problematic on
   103→   103→// mobile keyboards and the four rules above already defeat dictionary attacks
   104→   104→// when combined with 12+ length and bcrypt cost 12.)
   105→   105→
   106→   106→export const PASSWORD_MIN_LENGTH = 12;
   107→   107→
   108→   108→export function validatePasswordPolicy(password: string): {
   109→   109→  ok: boolean;
   110→   110→  error?: string;
   111→   111→} {
   112→   112→  if (password.length < PASSWORD_MIN_LENGTH) {
   113→   113→    return { ok: false, error: `Password must be at least ${PASSWORD_MIN_LENGTH} characters` };
   114→   114→  }
   115→   115→  if (!/[A-Z]/.test(password)) {
   116→   116→    return { ok: false, error: "Password must contain at least one uppercase letter" };
   117→   117→  }
   118→   118→  if (!/[a-z]/.test(password)) {
   119→   119→    return { ok: false, error: "Password must contain at least one lowercase letter" };
   120→   120→  }
   121→   121→  if (!/[0-9]/.test(password)) {
   122→   122→    return { ok: false, error: "Password must contain at least one digit" };
   123→   123→  }
   124→   124→  return { ok: true };
   125→   125→}

   167→   167→export async function signSession(payload: SessionPayload): Promise<string> {
   168→   168→  return new SignJWT({ email: payload.email, role: payload.role })
   169→   169→    .setProtectedHeader({ alg: "HS256" })
   170→   170→    .setSubject(payload.sub)
   171→   171→    .setIssuedAt()
   172→   172→    .setExpirationTime(`${SESSION_TTL_SECONDS}s`)
   173→   173→    .sign(JWT_SECRET);
   174→   174→}
   175→   175→
   176→   176→async function verifySession(token: string): Promise<SessionPayload | null> {
   177→   177→  try {
   178→   178→    const { payload } = await jwtVerify(token, JWT_SECRET, {
   179→   179→      algorithms: ["HS256"],
   180→   180→    });
   181→   181→    if (
   182→   182→      typeof payload.sub !== "string" ||
   183→   183→      typeof payload.email !== "string" ||
   184→   184→      typeof payload.role !== "string"
   185→   185→    ) {
   186→   186→      return null;
   187→   187→    }
   188→   188→    return {
   189→   189→      sub: payload.sub,
   190→   190→      email: payload.email,
   191→   191→      role: payload.role,
   192→   192→      iat: payload.iat,
   193→   193→      exp: payload.exp,
   194→   194→    };
   195→   195→  } catch {
   196→   196→    return null;
   197→   197→  }
   198→   198→}

   218→   218→// Reads session from cookies(). Works in Server Components and Route Handlers.
   219→   219→// Does NOT work in middleware (use verifySessionFromRequest instead).
   220→   220→export async function getSession(): Promise<SessionPayload | null> {
   221→   221→  const store = await cookies();
   222→   222→  const token = store.get(SESSION_COOKIE)?.value;
   223→   223→  if (!token) return null;
   224→   224→  return verifySession(token);
   225→   225→}
   226→   226→
   227→   227→// Reads session from a raw cookie header string (for middleware use).
   228→   228→export async function getSessionFromCookieHeader(
   229→   229→  cookieHeader: string | null | undefined,
   230→   230→): Promise<SessionPayload | null> {
   231→   231→  if (!cookieHeader) return null;
   232→   232→  const cookies = Object.fromEntries(
   233→   233→    cookieHeader
   234→   234→      .split(";")
   235→   235→      .map((c) => c.trim().split("="))
   236→   236→      .filter(([k]) => k)
   237→   237→      .map(([k, ...rest]) => [k, decodeURIComponent(rest.join("="))]),
   238→   238→  );
   239→   239→  const token = cookies[SESSION_COOKIE];
   240→   240→  if (!token) return null;
   241→   241→  return verifySession(token);
   242→   242→}

   268→   268→      email: user.email,
   269→   269→      role: user.role,

   273→   273→  } catch (err) {

   319→   319→// ---- TOTP 2FA --------------------------------------------------------------
   320→   320→// RFC 6238 TOTP using SHA-1, 30s step, 6 digits — compatible with Google
   321→   321→// Authenticator, Authy, 1Password, Bitwarden, Microsoft Authenticator, etc.
   322→   322→// ============================================================================
   323→   323→
   324→   324→export interface TOTPSecret {
   325→   325→  base32: string;          // store this in DB
   326→   326→  otpauthUrl: string;      // for the QR code
   327→   327→  qrCodeDataUrl: string;   // ready to render in <img src=...>
   328→   328→}
   329→   329→
   330→   330→export async function generateTOTPSecret(userEmail: string): Promise<TOTPSecret> {
   331→   331→  const secret = new OTPAuth.Secret({ size: 20 });
   332→   332→
   333→   333→  const totp = new OTPAuth.TOTP({
   334→   334→    issuer: "Planned",
   335→   335→    label: userEmail,
   336→   336→    algorithm: "SHA1",
   337→   337→    digits: 6,
   338→   338→    period: 30,
   339→   339→    secret,
   340→   340→  });
   341→   341→
   342→   342→  const otpauthUrl = totp.toString();
   343→   343→  const qrCodeDataUrl = await QRCode.toDataURL(otpauthUrl, {
   344→   344→    margin: 1,
   345→   345→    width: 240,
   346→   346→    color: { dark: "#0A4D5C", light: "#FFFFFF" },
   347→   347→  });
   348→   348→
   349→   349→  return {
   350→   350→    base32: secret.base32,
   351→   351→    otpauthUrl,
   352→   352→    qrCodeDataUrl,
   353→   353→  };
   354→   354→}

   356→   356→export function verifyTOTP(token: string, base32Secret: string): boolean {
   357→   357→  if (!/^\d{6}$/.test(token)) return false;
   358→   358→  const secret = OTPAuth.Secret.fromBase32(base32Secret);
   359→   359→  const totp = new OTPAuth.TOTP({
   360→   360→    issuer: "Planned",
   361→   361→    label: "",
   362→   362→    algorithm: "SHA1",
   363→   363→    digits: 6,
   364→   364→    period: 30,
   365→   365→    secret,
   366→   366→  });
   367→   367→  // otpauth returns 0 for invalid; we want a boolean.
   368→   368→  // window = TOTP_WINDOW allows ±1 step (30s each direction) for clock drift.
   369→   369→  const delta = totp.validate({ token, window: TOTP_WINDOW });
   370→   370→  return delta !== null;
   371→   371→}

   373→   373→// Generate N single-use backup codes. Return both plaintext (to show user once)
   374→   374→// and bcrypt hashes (to store). Each plaintext code is 8 chars, dashed:
   375→   375→// e.g. "ABCD-1234-EFGH"
   376→   376→export async function generateBackupCodes(): Promise<{
   377→   377→  plaintext: string[];
   378→   378→  hashes: string[];
   379→   379→}> {
   380→   380→  const charset = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // no ambiguous chars
   381→   381→  const plaintext: string[] = [];
   382→   382→  for (let i = 0; i < BACKUP_CODE_COUNT; i++) {
   383→   383→    const chars = Array.from(
   384→   384→      { length: 12 },
   385→   385→      () => charset[Math.floor(Math.random() * charset.length)],
   386→   386→    ).join("");
   387→   387→    plaintext.push(`${chars.slice(0, 4)}-${chars.slice(4, 8)}-${chars.slice(8, 12)}`);
   388→   388→  }
   389→   389→  const hashes = await Promise.all(plaintext.map((c) => bcrypt.hash(c, BCRYPT_COST)));
   390→   390→  return { plaintext, hashes };
   391→   391→}
   392→   392→
   393→   393→// Verify a backup code against the stored hash array, returning the index of
   394→   394→// the matched code (so the caller can remove it from the array).
   395→   395→export async function verifyBackupCode(
   396→   396→  candidate: string,
   397→   397→  hashesJson: string,
   398→   398→): Promise<number | null> {
   399→   399→  let hashes: string[] = [];
   400→   400→  try {
   401→   401→    hashes = JSON.parse(hashesJson);
   402→   402→  } catch {
   403→   403→    return null;
   404→   404→  }
   405→   405→  const normalized = candidate.trim().toUpperCase();
   406→   406→  for (let i = 0; i < hashes.length; i++) {
   407→   407→    const ok = await bcrypt.compare(normalized, hashes[i]);
   408→   408→    if (ok) return i;
   409→   409→  }
   410→   410→  return null;
   411→   411→}

   413→   413→// ---- Safe user registration (NEVER trusts client role) --------------------
   414→   414→// Use this for ANY future public registration endpoint. It hardcodes role=USER
   415→   415→// and explicitly ignores any incoming `role` field. This is the only safe way
   416→   416→// to create non-founder accounts.

   419→   419→export interface RegisterInput {
   420→   420→  email: string;
   421→   421→  password: string;
   422→   422→  name: string;
   423→   423→  // NOTE: deliberately no `role` field. Even if a caller passes role, we ignore.
   424→   424→}
   425→   425→
   426→   426→export async function registerUserAsUSER(input: RegisterInput): Promise<{
   427→   427→  ok: boolean;
   428→   428→  error?: string;
   429→   429→  userId?: string;
   430→   430→}> {
   431→   431→  const email = input.email.trim().toLowerCase();
   432→   432→  if (!email || !email.includes("@")) {
   433→   433→    return { ok: false, error: "Valid email required" };
   434→   434→  }
   435→   435→  const passwordCheck = validatePasswordPolicy(input.password);
   436→   436→  if (!passwordCheck.ok) {
   437→   437→    return { ok: false, error: passwordCheck.error };
   438→   438→  }
   439→   439→  if (input.name.trim().length < 2) {
   440→   440→    return { ok: false, error: "Name required" };
   441→   441→  }
   442→   442→
   443→   443→  // Check email not in use
   444→   444→  const existing = await db.user.findUnique({ where: { email } });
   445→   445→  if (existing) {
   446→   446→    return { ok: false, error: "Email already registered" };
   447→   447→  }
   448→   448→
   449→   449→  const passwordHash = await hashPassword(input.password);
   450→   450→
   451→   451→  // CRITICAL: role is hardcoded to USER_ROLE. NEVER accept this from the client.
   452→   452→  // Even if a future caller passes a role, we ignore it.
   453→   453→  const user = await db.user.create({
   454→   454→    data: {
   455→   455→      email,
   456→   456→      name: input.name.trim(),
   457→   457→      passwordHash,
   458→   458→      role: USER_ROLE, // <-- hardcoded, non-overridable
   459→   459→    },
   460→   460→  });
   461→   461→
   462→   462→  await auditLog({
   463→   463→    userId: user.id,
   464→   464→    action: "USER_REGISTERED",
   465→   465→    entityType: "user",
   466→   466→    entityId: user.id,
   467→   467→    after: { email: user.email, role: user.role },
   468→   468→    success: true,
   469→   469→  });
   470→   470→
   471→   471→  return { ok: true, userId: user.id };
   472→   472→}

   476→   476→export interface LoginResult {
   477→   477→  ok: boolean;

   485→   485→
   486→   486→// In-memory store of pending 2FA challenges. For production at scale, use
   487→   487→// Redis or DB-backed sessions. For Planned's single-admin scope, this is fine.
   488→   488→const twoFactorChallenges = new Map<
   489→   489→  string,
   490→   490→  { userId: string; expiresAt: number }
   491→   491→>();
   492→   492→const TWO_FACTOR_CHALLENGE_TTL_MS = 5 * 60 * 1000;
   493→   493→
   494→   494→function createTwoFactorChallenge(userId: string): string {
   495→   495→  const token = crypto.randomUUID() + crypto.randomUUID();
   496→   496→  twoFactorChallenges.set(token, {
   497→   497→    userId,
   498→   498→    expiresAt: Date.now() + TWO_FACTOR_CHALLENGE_TTL_MS,
   499→   499→  });
   500→   500→  return token;
   501→   501→}
   502→   502→
   503→   503→function consumeTwoFactorChallenge(token: string): string | null {
   504→   504→  const entry = twoFactorChallenges.get(token);
   505→   505→  if (!entry) return null;
   506→   506→  twoFactorChallenges.delete(token); // single-use
   507→   507→  if (Date.now() > entry.expiresAt) return null;
   508→   508→  return entry.userId;
   509→   509→}

   511→   511→export async function attemptLogin(params: {
   512→   512→  email: string;
   513→   513→  password: string;
   514→   514→  ipAddress?: string;
   515→   515→  userAgent?: string;
   516→   516→}): Promise<LoginResult> {
   517→   517→  const email = params.email.trim().toLowerCase();
   518→   518→
   519→   519→  try {
   520→   520→    const user = await db.user.findUnique({
   521→   521→      where: { email },
   522→   522→    });
   523→   523→
   524→   524→    if (!user || !user.passwordHash || user.deletedAt) {
   525→   525→      // Don't reveal whether email exists — log and fail uniformly.
   526→   526→      await auditLog({
   527→   527→        action: "LOGIN_FAILED",
   528→   528→        entityType: "user",
   529→   529→        entityId: email,
   530→   530→        ipAddress: params.ipAddress,
   531→   531→        userAgent: params.userAgent,
   532→   532→        success: false,
   533→   533→      });
   534→   534→      return { ok: false, error: "Invalid email or password" };
   535→   535→    }
   536→   536→
   537→   537→    // Check lockout
   538→   538→    if (user.lockedUntil && user.lockedUntil > new Date()) {
   539→   539→      await auditLog({
   540→   540→        userId: user.id,
   541→   541→        action: "LOGIN_LOCKED",
   542→   542→        entityType: "user",
   543→   543→        entityId: user.id,
   544→   544→        ipAddress: params.ipAddress,
   545→   545→        userAgent: params.userAgent,
   546→   546→        success: false,
   547→   547→      });
   548→   548→      return {
   549→   549→        ok: false,
   550→   550→        error: "Account locked. Try again later or use security question recovery.",
   551→   551→      };
   552→   552→    }
   553→   553→
   554→   554→    const passwordOk = await verifyPassword(params.password, user.passwordHash);
   555→   555→    if (!passwordOk) {
   556→   556→      const newAttempts = user.failedLoginAttempts + 1;
   557→   557→      const shouldLock = newAttempts >= MAX_LOGIN_ATTEMPTS;
   558→   558→      await db.user.update({
   559→   559→        where: { id: user.id },
   560→   560→        data: {
   561→   561→          failedLoginAttempts: newAttempts,
   562→   562→          lockedUntil: shouldLock
   563→   563→            ? new Date(Date.now() + LOCKOUT_MINUTES * 60 * 1000)
   564→   564→            : null,
   565→   565→        },
   566→   566→      });
   567→   567→      await auditLog({
   568→   568→        userId: user.id,
   569→   569→        action: "LOGIN_FAILED",
   570→   570→        entityType: "user",
   571→   571→        entityId: user.id,
   572→   572→        ipAddress: params.ipAddress,
   573→   573→        userAgent: params.userAgent,
   574→   574→        success: false,
   575→   575→        after: { attempts: newAttempts, locked: shouldLock },
   576→   576→      });
   577→   577→      return { ok: false, error: "Invalid email or password" };
   578→   578→    }
   579→   579→
   580→   580→    // Password OK — if 2FA enabled, issue a challenge instead of session
   581→   581→    if (user.twoFactorEnabled && user.twoFactorSecret) {
   582→   582→      const challenge = createTwoFactorChallenge(user.id);
   583→   583→      await auditLog({
   584→   584→        userId: user.id,
   585→   585→        action: "LOGIN_2FA_CHALLENGE_ISSUED",
   586→   586→        entityType: "user",
   587→   587→        entityId: user.id,
   588→   588→        ipAddress: params.ipAddress,
   589→   589→        userAgent: params.userAgent,
   590→   590→        success: true,
   591→   591→      });
   592→   592→      return {
   593→   593→        ok: false,
   594→   594→        twoFactorRequired: true,
   595→   595→        twoFactorChallenge: challenge,
   596→   596→        error: "Two-factor authentication required",
   597→   597→      };
   598→   598→    }
   599→   599→
   600→   600→    // No 2FA — reset counters and issue session
   601→   601→    await db.user.update({
   602→   602→      where: { id: user.id },
   603→   603→      data: {
   604→   604→        failedLoginAttempts: 0,
   605→   605→        lockedUntil: null,
   606→   606→        lastLoginAt: new Date(),
   607→   607→      },
   608→   608→    });
   609→   609→
   610→   610→    const token = await signSession({
   611→   611→      sub: user.id,
   612→   612→      email: user.email,
   613→   613→      role: user.role,
   614→   614→    });
   615→   615→    await setSessionCookie(token);
   616→   616→
   617→   617→    await auditLog({
   618→   618→      userId: user.id,
   619→   619→      action: "LOGIN_SUCCESS",
   620→   620→      entityType: "user",
   621→   621→      entityId: user.id,
   622→   622→      ipAddress: params.ipAddress,
   623→   623→      userAgent: params.userAgent,
   624→   624→      success: true,
   625→   625→    });
   626→   626→
   627→   627→    return {
   628→   628→      ok: true,
   629→   629→      user: { id: user.id, email: user.email, role: user.role, name: user.name, twoFactorEnabled: user.twoFactorEnabled },
   630→   630→    };
   631→   631→  } catch (err) {
   632→   632→    logger.error("Login error", { err, email });
   633→   633→    return { ok: false, error: "An unexpected error occurred" };
   634→   634→  }
   635→   635→}

   639→   639→export interface Complete2FAResult {
   640→   640→  ok: boolean;
   641→   641→  error?: string;
   642→   642→  user?: AuthUser;
   643→   643→}
   644→   644→
   645→   645→export async function completeLoginWith2FA(params: {
   646→   646→  challenge: string;
   647→   647→  code: string;
   648→   648→  ipAddress?: string;
   649→   649→  userAgent?: string;
   650→   650→}): Promise<Complete2FAResult> {
   651→   651→  try {
   652→   652→    const userId = consumeTwoFactorChallenge(params.challenge);
   653→   653→    if (!userId) {
   654→   654→      return { ok: false, error: "Invalid or expired 2FA challenge" };
   655→   655→    }
   656→   656→
   657→   657→    const user = await db.user.findUnique({
   658→   658→      where: { id: userId, deletedAt: null },
   659→   659→    });
   660→   660→    if (!user || !user.twoFactorEnabled || !user.twoFactorSecret) {
   661→   661→      return { ok: false, error: "2FA not enabled for this account" };
   662→   662→    }
   663→   663→
   664→   664→    // Try TOTP first
   665→   665→    let verified = false;
   666→   666→    let usedBackupIndex: number | null = null;
   667→   667→
   668→   668→    if (verifyTOTP(params.code, user.twoFactorSecret)) {
   669→   669→      verified = true;
   670→   670→    } else if (user.twoFactorBackupCodesHash) {
   671→   671→      // Try backup code
   672→   672→      usedBackupIndex = await verifyBackupCode(
   673→   673→        params.code,
   674→   674→        user.twoFactorBackupCodesHash,
   675→   675→      );
   676→   676→      if (usedBackupIndex !== null) {
   677→   677→        verified = true;
   678→   678→        // Remove the used backup code from the array
   679→   679→        const hashes: string[] = JSON.parse(user.twoFactorBackupCodesHash);
   680→   680→        hashes.splice(usedBackupIndex, 1);
   681→   681→        await db.user.update({
   682→   682→          where: { id: user.id },
   683→   683→          data: { twoFactorBackupCodesHash: JSON.stringify(hashes) },
   684→   684→        });
   685→   685→      }
   686→   686→    }
   687→   687→
   688→   688→    if (!verified) {
   689→   689→      await auditLog({
   690→   690→        userId: user.id,
   691→   691→        action: "LOGIN_2FA_FAILED",
   692→   692→        entityType: "user",
   693→   693→        entityId: user.id,
   694→   694→        ipAddress: params.ipAddress,
   695→   695→        userAgent: params.userAgent,
   696→   696→        success: false,
   697→   697→      });
   698→   698→      return { ok: false, error: "Invalid 2FA code" };
   699→   699→    }
   700→   700→
   701→   701→    // Success — reset counters and issue session
   702→   702→    await db.user.update({
   703→   703→      where: { id: user.id },
   704→   704→      data: {
   705→   705→        failedLoginAttempts: 0,
   706→   706→        lockedUntil: null,
   707→   707→        lastLoginAt: new Date(),
   708→   708→      },
   709→   709→    });
   710→   710→
   711→   711→    const token = await signSession({
   712→   712→      sub: user.id,
   713→   713→      email: user.email,
   714→   714→      role: user.role,
   715→   715→    });
   716→   716→    await setSessionCookie(token);
   717→   717→
   718→   718→    await auditLog({
   719→   719→      userId: user.id,
   720→   720→      action: usedBackupIndex !== null ? "LOGIN_2FA_BACKUP_SUCCESS" : "LOGIN_2FA_SUCCESS",
   721→   721→      entityType: "user",
   722→   722→      entityId: user.id,
   723→   723→      ipAddress: params.ipAddress,
   724→   724→      userAgent: params.userAgent,
   725→   725→      success: true,
   726→   726→      after: { backupCodesRemaining: usedBackupIndex !== null },
   727→   727→    });
   728→   728→
   729→   729→    return {
   730→   730→      ok: true,
   731→   731→      user: { id: user.id, email: user.email, role: user.role, name: user.name, twoFactorEnabled: user.twoFactorEnabled },
   732→   732→    };
   733→   733→  } catch (err) {
   734→   734→    logger.error("2FA complete error", { err });
   735→   735→    return { ok: false, error: "An unexpected error occurred" };
   736→   736→  }
   737→   737→}

   787→   787→export async function performFounderSetup(params: {
   788→   788→  email: string;
   789→   789→  password: string;
   790→   790→  name: string;
   791→   791→  securityQuestion: string;
   792→   792→  securityAnswer: string;

   794→   794→  totpSecret?: string;        // base32 secret (from generateTOTPSecret)
   795→   795→  totpVerificationCode?: string; // 6-digit code to confirm enrollment
   796→   796→  ipAddress?: string;
   797→   797→  userAgent?: string;
   798→   798→}): Promise<SetupResult> {
   799→   799→  const email = params.email.trim().toLowerCase();
   800→   800→
   801→   801→  // Validate inputs
   802→   802→  if (!email || !email.includes("@")) {
   803→   803→    return { ok: false, error: "A valid email is required" };
   804→   804→  }
   805→   805→  const passwordCheck = validatePasswordPolicy(params.password);
   806→   806→  if (!passwordCheck.ok) {
   807→   807→    return { ok: false, error: passwordCheck.error };
   808→   808→  }
   809→   809→  if (params.name.trim().length < 2) {
   810→   810→    return { ok: false, error: "Name is required" };
   811→   811→  }
   812→   812→  if (params.securityQuestion.trim().length < 10) {
   813→   813→    return { ok: false, error: "Security question must be at least 10 characters" };
   814→   814→  }
   815→   815→  if (params.securityAnswer.trim().length < 2) {
   816→   816→    return { ok: false, error: "Security answer is required" };
   817→   817→  }
   818→   818→
   819→   819→  // CRITICAL: refuse if already initialized
   820→   820→  const alreadyInitialized = await isSystemInitialized();
   821→   821→  if (alreadyInitialized) {
   822→   822→    await auditLog({
   823→   823→      action: "SETUP_REFUSED_ALREADY_INITIALIZED",
   824→   824→      entityType: "system",
   825→   825→      entityId: "singleton",
   826→   826→      ipAddress: params.ipAddress,
   827→   827→      userAgent: params.userAgent,
   828→   828→      success: false,
   829→   829→    });
   830→   830→    return { ok: false, error: "System is already initialized" };
   831→   831→  }
   832→   832→
   833→   833→  // Refuse if any SUPER_ADMIN already exists (defensive — shouldn't happen)
   834→   834→  try {
   835→   835→    const existing = await db.user.findFirst({
   836→   836→      where: { role: SUPER_ADMIN_ROLE, deletedAt: null },
   837→   837→    });
   838→   838→    if (existing) {
   839→   839→      return {
   840→   840→        ok: false,
   841→   841→        error: "A super admin already exists. Setup is no longer available.",
   842→   842→      };
   843→   843→    }
   844→   844→  } catch (err) {
   845→   845→    logger.error("Setup: failed to check existing super admin", { err });
   846→   846→    return { ok: false, error: "Setup check failed" };
   847→   847→  }
   848→   848→
   849→   849→  // Validate TOTP enrollment if provided
   850→   850→  let enrollTotp = false;
   851→   851→  if (params.totpSecret && params.totpVerificationCode) {
   852→   852→    if (!verifyTOTP(params.totpVerificationCode, params.totpSecret)) {
   853→   853→      return { ok: false, error: "Invalid 2FA verification code. Please try again." };
   854→   854→    }
   855→   855→    enrollTotp = true;
   856→   856→  }
   857→   857→
   858→   858→  // Hash credentials
   859→   859→  const [passwordHash, securityAnswerHash] = await Promise.all([
   860→   860→    hashPassword(params.password),
   861→   861→    hashSecurityAnswer(params.securityAnswer),
   862→   862→  ]);
   863→   863→
   864→   864→  // Generate backup codes if enrolling in 2FA
   865→   865→  let backupCodesHash: string | null = null;
   866→   866→  let backupCodesPlaintext: string[] | null = null;
   867→   867→  if (enrollTotp) {
   868→   868→    const { plaintext, hashes } = await generateBackupCodes();
   869→   869→    backupCodesHash = JSON.stringify(hashes);
   870→   870→    backupCodesPlaintext = plaintext;
   871→   871→  }
   872→   872→
   873→   873→  // Create the founder — role HARDCODED to SUPER_ADMIN, not from client.
   874→   874→  // This is the ONE place SUPER_ADMIN is set; everywhere else uses USER_ROLE.
   875→   875→  const founder = await db.user.create({
   876→   876→    data: {
   877→   877→      email,
   878→   878→      name: params.name.trim(),
   879→   879→      passwordHash,
   880→   880→      role: SUPER_ADMIN_ROLE, // <-- hardcoded, the only SUPER_ADMIN creation path
   881→   881→      securityQuestion: params.securityQuestion.trim(),
   882→   882→      securityAnswerHash,
   883→   883→      emailVerified: new Date(),
   884→   884→      lastLoginAt: new Date(),
   885→   885→      twoFactorSecret: params.totpSecret ?? null,
   886→   886→      twoFactorEnabled: enrollTotp,
   887→   887→      twoFactorEnrolledAt: enrollTotp ? new Date() : null,
   888→   888→      twoFactorBackupCodesHash: backupCodesHash,
   889→   889→    },
   890→   890→  });
   891→   891→
   892→   892→  // Mark system as initialized — superAdminUserId is AUTHORITATIVE.
   893→   893→  // superAdminEmail is metadata only (mirror of User.email at setup time).
   894→   894→  await db.systemSettings.upsert({
   895→   895→    where: { id: "singleton" },
   896→   896→    update: {
   897→   897→      isInitialized: true,
   898→   898→      superAdminUserId: founder.id, // <-- AUTHORITATIVE
   899→   899→      superAdminEmail: founder.email, // <-- metadata only
   900→   900→      setupCompletedAt: new Date(),
   901→   901→    },
   902→   902→    create: {
   903→   903→      id: "singleton",
   904→   904→      isInitialized: true,
   905→   905→      superAdminUserId: founder.id,
   906→   906→      superAdminEmail: founder.email,
   907→   907→      setupCompletedAt: new Date(),
   908→   908→    },
   909→   909→  });
   910→   910→
   911→   911→  // Issue session
   912→   912→  const token = await signSession({
   913→   913→    sub: founder.id,
   914→   914→    email: founder.email,
   915→   915→    role: founder.role,
   916→   916→  });
   917→   917→  await setSessionCookie(token);
   918→   918→
   919→   919→  await auditLog({
   920→   920→    userId: founder.id,
   921→   921→    action: "FOUNDER_SETUP_COMPLETED",
   922→   922→    entityType: "system",
   923→   923→    entityId: "singleton",
   924→   924→    after: {
   925→   925→      email: founder.email,
   926→   926→      name: founder.name,
   927→   927→      userId: founder.id, // log the authoritative ID
   928→   928→      twoFactorEnabled: enrollTotp,
   929→   929→    },
   930→   930→    ipAddress: params.ipAddress,
   931→   931→    userAgent: params.userAgent,
   932→   932→    success: true,
   933→   933→  });
   934→   934→
   935→   935→  return {
   936→   936→    ok: true,
   937→   937→    user: {
   938→   938→      id: founder.id,
   939→   939→      email: founder.email,
   940→   940→      role: founder.role,
   941→   941→      name: founder.name,
   942→   942→      twoFactorEnabled: enrollTotp,
   943→   943→    },
   944→   944→  };
   945→   945→}

   984→   984→
   985→   985→    if (!verifyTOTP(params.totpVerificationCode, params.totpSecret)) {
   986→   986→      return { ok: false, error: "Invalid 2FA verification code" };
   987→   987→    }
   988→   988→
   989→   989→    const { plaintext, hashes } = await generateBackupCodes();
   990→   990→

  1018→  1018→// ---- Change password (founder-initiated reset) -----------------------------
  1019→  1019→// The founder can change their own password from /admin/security. Requires:
  1020→  1020→//   - current password (proves identity)
  1021→  1021→//   - new password (must pass password policy)
  1022→  1022→//   - 2FA code (if 2FA is enabled — defense in depth)
  1023→  1023→//
  1024→  1024→// CRITICAL: this NEVER changes the user ID. The founder account remains
  1025→  1025→// bound to the same user ID forever — only the passwordHash field updates.
  1026→  1026→
  1027→  1027→export async function changePassword(params: {
  1028→  1028→  userId: string;
  1029→  1029→  currentPassword: string;
  1030→  1030→  newPassword: string;
  1031→  1031→  totpCode?: string;
  1032→  1032→  ipAddress?: string;
  1033→  1033→  userAgent?: string;
  1034→  1034→}): Promise<{ ok: boolean; error?: string }> {
  1035→  1035→  try {
  1036→  1036→    const user = await db.user.findUnique({
  1037→  1037→      where: { id: params.userId, deletedAt: null },
  1038→  1038→    });
  1039→  1039→    if (!user || !user.passwordHash) {
  1040→  1040→      return { ok: false, error: "Account not found" };
  1041→  1041→    }
  1042→  1042→
  1043→  1043→    // Verify current password
  1044→  1044→    const passwordOk = await verifyPassword(params.currentPassword, user.passwordHash);
  1045→  1045→    if (!passwordOk) {
  1046→  1046→      await auditLog({
  1047→  1047→        userId: params.userId,
  1048→  1048→        action: "PASSWORD_CHANGE_WRONG_PASSWORD",
  1049→  1049→        entityType: "user",
  1050→  1050→        entityId: params.userId,
  1051→  1051→        ipAddress: params.ipAddress,
  1052→  1052→        userAgent: params.userAgent,
  1053→  1053→        success: false,
  1054→  1054→      });
  1055→  1055→      return { ok: false, error: "Current password is incorrect" };
  1056→  1056→    }
  1057→  1057→
  1058→  1058→    // If 2FA enabled, require TOTP code
  1059→  1059→    if (user.twoFactorEnabled && user.twoFactorSecret) {
  1060→  1060→      if (!params.totpCode) {
  1061→  1061→        return { ok: false, error: "2FA code is required to change password" };
  1062→  1062→      }
  1063→  1063→      let twoFactorOk = verifyTOTP(params.totpCode, user.twoFactorSecret);
  1064→  1064→      if (!twoFactorOk && user.twoFactorBackupCodesHash) {
  1065→  1065→        const idx = await verifyBackupCode(params.totpCode, user.twoFactorBackupCodesHash);
  1066→  1066→        if (idx !== null) {
  1067→  1067→          twoFactorOk = true;
  1068→  1068→          const hashes: string[] = JSON.parse(user.twoFactorBackupCodesHash);
  1069→  1069→          hashes.splice(idx, 1);
  1070→  1070→          await db.user.update({
  1071→  1071→            where: { id: params.userId },
  1072→  1072→            data: { twoFactorBackupCodesHash: JSON.stringify(hashes) },
  1073→  1073→          });
  1074→  1074→        }
  1075→  1075→      }
  1076→  1076→      if (!twoFactorOk) {
  1077→  1077→        await auditLog({
  1078→  1078→          userId: params.userId,
  1079→  1079→          action: "PASSWORD_CHANGE_2FA_FAILED",
  1080→  1080→          entityType: "user",
  1081→  1081→          entityId: params.userId,
  1082→  1082→          ipAddress: params.ipAddress,
  1083→  1083→          userAgent: params.userAgent,
  1084→  1084→          success: false,
  1085→  1085→        });
  1086→  1086→        return { ok: false, error: "Invalid 2FA code" };
  1087→  1087→      }
  1088→  1088→    }
  1089→  1089→
  1090→  1090→    // Validate new password against policy
  1091→  1091→    const policyCheck = validatePasswordPolicy(params.newPassword);
  1092→  1092→    if (!policyCheck.ok) {
  1093→  1093→      return { ok: false, error: policyCheck.error };
  1094→  1094→    }
  1095→  1095→
  1096→  1096→    // Refuse if new password equals current password
  1097→  1097→    if (params.newPassword === params.currentPassword) {
  1098→  1098→      return { ok: false, error: "New password must be different from current password" };
  1099→  1099→    }
  1100→  1100→
  1101→  1101→    // Apply — NOTE: only passwordHash updates. User ID, email, role, 2FA
  1102→  1102→    // enrollment, and all other identity fields are untouched.
  1103→  1103→    const newHash = await hashPassword(params.newPassword);
  1104→  1104→    await db.user.update({
  1105→  1105→      where: { id: params.userId },
  1106→  1106→      data: { passwordHash: newHash },
  1107→  1107→    });
  1108→  1108→
  1109→  1109→    await auditLog({
  1110→  1110→      userId: params.userId,
  1111→  1111→      action: "PASSWORD_CHANGED",
  1112→  1112→      entityType: "user",
  1113→  1113→      entityId: params.userId,
  1114→  1114→      ipAddress: params.ipAddress,
  1115→  1115→      userAgent: params.userAgent,
  1116→  1116→      success: true,
  1117→  1117→      // Don't log passwords — even hashed. Just confirm the action.
  1118→  1118→      after: { policy: "12+ chars, upper+lower+digit", userIdPreserved: params.userId },
  1119→  1119→    });
  1120→  1120→
  1121→  1121→    return { ok: true };
  1122→  1122→  } catch (err) {
  1123→  1123→    logger.error("Change password error", { err });
  1124→  1124→    return { ok: false, error: "Failed to change password" };
  1125→  1125→  }
  1126→  1126→}
