   218→function startOfPeriod(cadence: GoalCadence, now: number = Date.now()): number {
   219→  const d = new Date(now);
   220→  if (cadence === "weekly") {
   221→    const day = d.getDay(); // 0 = Sun
   222→    d.setDate(d.getDate() - day);
   223→    d.setHours(0, 0, 0, 0);
   224→    return d.getTime();
   225→  }
   226→  if (cadence === "monthly") {
   227→    return new Date(d.getFullYear(), d.getMonth(), 1).getTime();
   228→  }
   229→  // annual
   230→  return new Date(d.getFullYear(), 0, 1).getTime();
   231→}

   401→  addGoal: (goal) => {

   432→    })),

   438→          ? {
   439→              ...g,
   440→              periodStart: startOfPeriod(g.cadence),
   441→              currentAmount: g.type === "spend_less" ? 0 : g.currentAmount,
   442→            }
   443→          : g
   444→      ),
   445→    })),
   446→
   447→  addTransaction: (tx) =>
   448→    set((s) => {
   449→      const newTx: Transaction = {
   450→        id: `tx-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   451→        timestamp: tx.timestamp ?? Date.now(),
   452→        ...tx,
   453→      };
   454→
   455→      // --------------------------------------------------------------------
   456→      // BUG #1 FIX: Save Money modal must update child.currentAmount.
   457→      // Previously the transaction was logged but the child's live savings
   458→      // balance never moved — money disappeared into the void.
   459→      // --------------------------------------------------------------------
   460→      let children = s.children;
   461→      let accounts = s.accounts;
   462→      let investments = s.investments;
   463→
   464→      if (newTx.type === "save") {
   465→        // Credit child's savings; debit linked external account.
   466→        children = children.map((c) =>
   467→          c.id === newTx.childId ? { ...c, currentAmount: c.currentAmount + newTx.amount } : c
   468→        );
   469→        if (newTx.accountId) {
   470→          accounts = accounts.map((a) =>
   471→            a.id === newTx.accountId ? { ...a, balance: Math.max(0, a.balance - newTx.amount) } : a
   472→          );
   473→        }
   474→      } else if (newTx.type === "withdraw") {
   475→        // Debit child's savings; credit external account.
   476→        children = children.map((c) =>
   477→          c.id === newTx.childId ? { ...c, currentAmount: Math.max(0, c.currentAmount - newTx.amount) } : c
   478→        );
   479→        if (newTx.accountId) {
   480→          accounts = accounts.map((a) =>
   481→            a.id === newTx.accountId ? { ...a, balance: a.balance + newTx.amount } : a
   482→          );
   483→        }
   484→      } else if (newTx.type === "invest") {
   485→        // Debit savings; create investment record handled by investNow().
   486→        children = children.map((c) =>
   487→          c.id === newTx.childId ? { ...c, currentAmount: Math.max(0, c.currentAmount - newTx.amount) } : c
   488→        );
   489→      } else if (newTx.type === "redeem") {
   490→        // Credit savings by tokens * TOKEN_REDEEM_RATE.
   491→        const cash = newTx.tokenDelta * TOKEN_REDEEM_RATE;
   492→        children = children.map((c) =>
   493→          c.id === newTx.childId ? { ...c, currentAmount: c.currentAmount + cash } : c
   494→        );
   495→      }
   496→      // parent_give: no cash moves.
   497→
   498→      return {
   499→        ...s,
   500→        transactions: [newTx, ...s.transactions],
   501→        children,
   502→        accounts,
   503→        investments,
   504→      };
   505→    }),
   506→
   507→  addSpendingEntry: (e) =>
   508→    set((s) => {
   509→      const newE: SpendingEntry = {
   510→        id: `spend-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   511→        timestamp: e.timestamp ?? Date.now(),
   512→        ...e,
   513→      };
   514→      // --------------------------------------------------------------------
   515→      // BUG #12 FIX: adding a spending entry must update the category's
   516→      // displayed `spent`. Because `spent` is no longer stored (bug #7 fix),
   517→      // the category bars will always reflect real entries — but we still
   518→      // surface the entry through getChildCategories() at render time.
   519→      // --------------------------------------------------------------------
   520→      return { ...s, spending: [newE, ...s.spending] };
   521→    }),
   522→
   523→  giveTokens: (childId, tokens, note) =>
   524→    set((s) => {
   525→      const entry: TokenLedgerEntry = {
   526→        id: `tok-${Date.now()}`,
   527→        childId,
   528→        type: "parent_give",
   529→        tokens,
   530→        note,
   531→        timestamp: Date.now(),
   532→      };
   533→      // Also log a transaction so the parent dashboard's activity feed shows it.
   534→      const tx: Transaction = {
   535→        id: `tx-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   536→        childId,
   537→        type: "parent_give",
   538→        amount: 0,
   539→        tokenDelta: tokens,
   540→        note,
   541→        timestamp: Date.now(),
   542→      };
   543→      return {
   544→        ...s,
   545→        tokenLedger: [entry, ...s.tokenLedger],
   546→        transactions: [tx, ...s.transactions],
   547→      };
   548→    }),
   549→
   550→  redeemTokens: (childId, tokens) => {
   551→    const s = get();
   552→    const balance = childTokenBalance(childId)(s);
   553→    if (tokens > balance) return false;
   554→
   555→    const cash = tokens * TOKEN_REDEEM_RATE;
   556→    const ledger: TokenLedgerEntry = {
   557→      id: `tok-${Date.now()}`,
   558→      childId,
   559→      type: "redeem",
   560→      tokens,
   561→      note: `${tokens} tokens redeemed @ ${TOKEN_REDEEM_RATE}`,
   562→      timestamp: Date.now(),
   563→    };
   564→    set((st) => ({
   565→      ...st,
   566→      tokenLedger: [ledger, ...st.tokenLedger],
   567→      // Use addTransaction's redeem path so the child's currentAmount updates.
   568→      transactions: [
   569→        {
   570→          id: `tx-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   571→          childId,
   572→          type: "redeem",
   573→          amount: cash,
   574→          tokenDelta: tokens,
   575→          note: `${tokens} tokens redeemed @ ${TOKEN_REDEEM_RATE}`,
   576→          timestamp: Date.now(),
   577→        },
   578→        ...st.transactions,
   579→      ],
   580→      children: st.children.map((c) =>
   581→        c.id === childId ? { ...c, currentAmount: c.currentAmount + cash } : c
   582→      ),
   583→    }));
   584→    return true;
   585→  },
   586→
   587→  investNow: (childId, amount, name, type) => {
   588→    const s = get();
   589→    const child = s.children.find((c) => c.id === childId);
   590→    if (!child || amount > child.currentAmount) return false;
   591→
   592→    const inv: Investment = {
   593→      id: `inv-${Date.now()}`,
   594→      childId,
   595→      name,
   596→      type,
   597→      amountInvested: amount,
   598→      currentValue: amount, // starts at par
   599→      status: "active",
   600→      openedAt: Date.now(),
   601→    };
   602→    set((st) => ({
   603→      ...st,
   604→      investments: [inv, ...st.investments],
   605→      children: st.children.map((c) =>
   606→        c.id === childId ? { ...c, currentAmount: Math.max(0, c.currentAmount - amount) } : c
   607→      ),
   608→      transactions: [
   609→        {
   610→          id: `tx-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   611→          childId,
   612→          type: "invest",
   613→          amount,
   614→          tokenDelta: 0,
   615→          investmentId: inv.id,
   616→          note: `Invested in ${name}`,
   617→          timestamp: Date.now(),
   618→        },
   619→        ...st.transactions,
   620→      ],
   621→    }));
   622→    return true;
   623→  },
   624→
   625→  resetSeed: () =>

   684→// Bug #7 FIX: per-owner spending categories. spent is computed from THIS
   685→// owner's entries, not a shared static field. Works for both children and parents.
   686→export const getOwnerCategories = (ownerId: string): Sel<
   687→  { category: SpendingCategory; spent: number; remaining: number; pct: number }[]
   688→> => (s) => {
   689→  const monthStart = new Date();
   690→  monthStart.setDate(1);
   691→  monthStart.setHours(0, 0, 0, 0);
   692→
   693→  return s.categories.map((category) => {
   694→    const spent = s.spending
   695→      .filter(
   696→        (e) =>
   697→          e.ownerId === ownerId &&
   698→          e.category === category.name &&
   699→          e.timestamp >= monthStart.getTime()
   700→      )
   701→      .reduce((sum, e) => sum + e.amount, 0);
   702→    const remaining = Math.max(0, category.budget - spent);
   703→    const pct = category.budget > 0 ? Math.min(100, (spent / category.budget) * 100) : 0;
   704→    return { category, spent, remaining, pct };
   705→  });
   706→};

   726→// Per-owner breakdown for the family overview — returns array of
   727→// { ownerId, ownerKind, ownerName, total } sorted by total descending.
   728→export const familySpendingBreakdown: Sel<
   729→  { ownerId: string; ownerKind: "parent" | "child"; ownerName: string; total: number }[]
   730→> = (s) => {
   731→  const now = Date.now();
   732→  const map = new Map<string, { ownerKind: "parent" | "child"; ownerName: string; total: number }>();
   733→  for (const e of s.spending) {
   734→    if (!isThisMonth(e.timestamp)) continue;
   735→    const existing = map.get(e.ownerId);
   736→    if (existing) {
   737→      existing.total += e.amount;
   738→    } else {
   739→      map.set(e.ownerId, { ownerKind: e.ownerKind, ownerName: e.ownerName, total: e.amount });
   740→    }
   741→  }
   742→  return Array.from(map.entries())
   743→    .map(([ownerId, v]) => ({ ownerId, ...v }))
   744→    .sort((a, b) => b.total - a.total);
   745→};
